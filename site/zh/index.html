curl https://api.socrates.ai/v1/dialogue \
  -H "Authorization: Bearer $YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "message": "为什么极限会存在？",
    "context": { "topic": "limits" },
    "mode": "dialogue"
  }'
from socrates import Client

client = Client()
dialogue = client.dialogue.create(
  message="为什么极限会存在？",
  topic="limits",
  mode="dialogue"
)
print(dialogue.next_question)
const dialogue = await socrates.dialogue.create({
  message: "为什么极限会存在？",
  context: { topic: "limits" },
  mode: "dialogue"
});

console.log(dialogue.nextQuestion);
